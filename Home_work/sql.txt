

CREATE TABLE IF NOT EXISTS talabalar (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    ism      TEXT NOT NULL,
    familiya TEXT NOT NULL,
    yosh     INTEGER NOT NULL,
    kurs     INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS oqituvchilar (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    ism      TEXT NOT NULL,
    familiya TEXT NOT NULL,
    fan      TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS buyurtmalar (
    buyurtma_id INTEGER PRIMARY KEY AUTOINCREMENT,
    mahsulot    TEXT NOT NULL,
    miqdor      INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS jonatmalar (
    jonatma_id  INTEGER PRIMARY KEY AUTOINCREMENT,
    buyurtma_id INTEGER,
    mahsulot    TEXT NOT NULL,
    miqdor      INTEGER NOT NULL
);

SELECT ism, familiya, 'Talaba' AS tur
FROM talabalar
WHERE ism LIKE 'A%' OR ism LIKE 'E%' OR ism LIKE 'I%'
   OR ism LIKE 'O%' OR ism LIKE 'U%'
UNION
SELECT ism, familiya, 'Oqituvchi' AS tur
FROM oqituvchilar
WHERE ism LIKE 'A%' OR ism LIKE 'E%' OR ism LIKE 'I%'
   OR ism LIKE 'O%' OR ism LIKE 'U%';

SELECT * FROM (
    SELECT ism, familiya FROM talabalar
    UNION
    SELECT ism, familiya FROM oqituvchilar
) ORDER BY LENGTH(familiya), familiya;


SELECT ism, familiya, 'Talaba' AS manba
FROM talabalar
UNION
SELECT ism, familiya, 'Oqituvchi' AS manba
FROM oqituvchilar
ORDER BY manba, familiya;


SELECT id, ism, familiya FROM (
    SELECT id, ism, familiya FROM talabalar
    ORDER BY id DESC LIMIT 3
) t
UNION
SELECT id, ism, familiya FROM (
    SELECT id, ism, familiya FROM oqituvchilar
    ORDER BY id DESC LIMIT 3
) o
ORDER BY id DESC;


SELECT mahsulot, miqdor FROM buyurtmalar
INTERSECT
SELECT mahsulot, miqdor FROM jonatmalar;


SELECT mahsulot, miqdor FROM buyurtmalar WHERE miqdor >= 5
INTERSECT
SELECT mahsulot, miqdor FROM jonatmalar WHERE miqdor >= 5;

SELECT buyurtma_id FROM buyurtmalar
INTERSECT
SELECT buyurtma_id FROM jonatmalar;


SELECT * FROM (
    SELECT mahsulot, miqdor FROM jonatmalar
    INTERSECT
    SELECT mahsulot, miqdor FROM buyurtmalar
) ORDER BY mahsulot;


SELECT mahsulot FROM buyurtmalar
EXCEPT
SELECT mahsulot FROM jonatmalar;

SELECT * FROM (
    SELECT mahsulot FROM jonatmalar
    EXCEPT
    SELECT mahsulot FROM buyurtmalar
) ORDER BY mahsulot;

SELECT buyurtma_id FROM buyurtmalar
WHERE buyurtma_id >= 10
EXCEPT
SELECT buyurtma_id FROM jonatmalar
WHERE buyurtma_id IS NOT NULL;


SELECT * FROM (
    SELECT jonatma_id FROM jonatmalar WHERE miqdor < 5
    EXCEPT
    SELECT jonatma_id FROM jonatmalar
    WHERE buyurtma_id IN (SELECT buyurtma_id FROM buyurtmalar)
      AND miqdor < 5
) ORDER BY jonatma_id DESC;
